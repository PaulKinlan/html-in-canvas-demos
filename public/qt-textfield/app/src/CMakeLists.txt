cmake_minimum_required(VERSION 3.21)
project(qttf LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 OpenGL)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(qttf main.cpp domitem.cpp domitem.h)
qt_add_qml_module(qttf URI QtTf VERSION 1.0 QML_FILES Main.qml DomTextField.qml SOURCES domitem.cpp domitem.h)
target_link_libraries(qttf PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL)
target_link_options(qttf PRIVATE -sEXPORTED_FUNCTIONS=_main,_malloc,_free,_qttf_changed,_qttf_event,_qttf_probe,_qttf_command -sMAX_WEBGL_VERSION=2)
