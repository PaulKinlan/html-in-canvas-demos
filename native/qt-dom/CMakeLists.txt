cmake_minimum_required(VERSION 3.21)
project(qtdom LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 OpenGL)
qt_standard_project_setup(REQUIRES 6.8)
qt_add_executable(qtdom main.cpp domitem.cpp domitem.h)
qt_add_qml_module(qtdom URI QtDom VERSION 1.0 QML_FILES Main.qml SOURCES domitem.cpp domitem.h)
target_link_libraries(qtdom PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::OpenGL)
target_link_options(qtdom PRIVATE -sEXPORTED_FUNCTIONS=_main,_qtdom_changed -sMAX_WEBGL_VERSION=2)
